#include <Arduino.h>            
#include <micro_ros_platformio.h>
#include <stdio.h>
#include <rcl/rcl.h>
#include <rcl/error_handling.h>
#include <rclc/rclc.h>
#include <rclc/executor.h>
#include <std_msgs/msg/bool.h>

// 定義硬體腳位
#define PUMP_PIN 4  // 繼電器 1：接 6V 真空泵
#define VALVE_PIN 5 // 繼電器 2：接 6V 電磁閥

rcl_subscription_t subscriber;
rcl_publisher_t publisher;
std_msgs__msg__Bool sub_msg;
std_msgs__msg__Bool pub_msg;

rclc_executor_t executor;
rclc_support_t support;
rcl_allocator_t allocator;
rcl_node_t node;

void subscription_callback(const void * msgin)
{
  const std_msgs__msg__Bool * msg = (const std_msgs__msg__Bool *)msgin;
  
  if (msg->data == true) {
    // 收到 true：執行「吸取」動作
    digitalWrite(VALVE_PIN, LOW); // 先確保電磁閥關閉 (密封氣路)
    delay(50);                    // 短暫延遲，確保繼電器確實切換
    digitalWrite(PUMP_PIN, HIGH); // 啟動真空泵開始抽氣
    
    pub_msg.data = true;          // 設定回傳訊息為 true (表示已吸取)
  } else {
    // 收到 false：執行「釋放」動作
    digitalWrite(PUMP_PIN, LOW);  // 先關閉真空泵
    delay(50);                    // 短暫延遲
    digitalWrite(VALVE_PIN, HIGH);// 打開電磁閥破壞真空
    
    // 延遲 500 毫秒，讓空氣進入吸盤 (可依實際硬體調整)
    delay(500); 
    
    digitalWrite(VALVE_PIN, LOW); // 關閉電磁閥，恢復待命狀態
    
    pub_msg.data = false;         // 設定回傳訊息為 false (表示已釋放)
  }
  
  // 動作執行完畢後，發布狀態回饋給 ROS 2 主機
  rcl_publish(&publisher, &pub_msg, NULL);
}

void setup() {
  set_microros_transports(); 
  
  // 初始化硬體腳位，預設皆為關閉
  pinMode(PUMP_PIN, OUTPUT);
  pinMode(VALVE_PIN, OUTPUT);
  digitalWrite(PUMP_PIN, LOW); 
  digitalWrite(VALVE_PIN, LOW); 
  
  delay(2000);

  allocator = rcl_get_default_allocator();
  rclc_support_init(&support, 0, NULL, &allocator);
  
  rclc_node_init_default(&node, "suction_controller_node", "", &support);

  // 初始化訂閱者 (接收指令)
  rclc_subscription_init_default(
    &subscriber,
    &node,
    ROSIDL_GET_MSG_TYPE_SUPPORT(std_msgs, msg, Bool),
    "suction_command");

  // 初始化發布者 (回傳狀態)
  rclc_publisher_init_default(
    &publisher,
    &node,
    ROSIDL_GET_MSG_TYPE_SUPPORT(std_msgs, msg, Bool),
    "suction_status");

  rclc_executor_init(&executor, &support.context, 1, &allocator);
  rclc_executor_add_subscription(&executor, &subscriber, &sub_msg, &subscription_callback, ON_NEW_DATA);
}

void loop() {
  delay(10);
  rclc_executor_spin_some(&executor, RCL_MS_TO_NS(100));
}
